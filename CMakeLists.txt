cmake_minimum_required(VERSION 3.20)
project(BlitzRemesher VERSION 0.4.0 LANGUAGES C CXX)
include(CTest)
option(BLITZ_TOOLS "Build importer and corpus/research tools" ON)
option(BLITZ_RESEARCH "Build pinned external baselines" OFF)
option(BLITZ_SANITIZE "Address and undefined-behavior sanitizers" OFF)
option(BLITZ_FUZZ "Build libFuzzer targets (Clang)" OFF)
option(BLITZ_AVX2 "Enable runtime-dispatched AVX2 kernels" ON)
option(BLITZ_CUDA "Build optional neural CUDA inference and evaluator" OFF)
option(BLITZ_NEURAL_TRAIN "Build C++ LibTorch training tool (requires BLITZ_CUDA)" OFF)
if(BLITZ_FLOAT_QUADRICS OR BLITZ_FLOAT_COSTS OR (DEFINED BLITZ_PACKED_COVERAGE AND NOT BLITZ_PACKED_COVERAGE))
  message(FATAL_ERROR "Experimental precision variants were removed. Production uses packed coverage and double quadrics/costs. See research/precision/ARCHIVE.md to restore the experiments.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(BLITZ_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(blitz_eval src/evaluate.cpp src/render_cost.cpp)
target_include_directories(blitz_eval PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_definitions(blitz_eval PRIVATE $<$<BOOL:${BLITZ_AVX2}>:BLITZ_AVX2>)
add_library(blitzremesher src/mesh.cpp src/simplify.cpp src/chain.cpp src/selection.cpp src/components.cpp src/c_api.cpp src/neural_graph.cpp src/neural_numeric.cpp src/neural_action.cpp src/neural.cpp)
target_include_directories(blitzremesher PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(blitzremesher PUBLIC blitz_eval)
target_compile_features(blitzremesher PUBLIC cxx_std_20)
target_compile_features(blitz_eval PUBLIC cxx_std_20)
set_target_properties(blitzremesher blitz_eval PROPERTIES POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
set_target_properties(blitzremesher blitz_eval PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 4)
set_target_properties(blitzremesher PROPERTIES EXPORT_NAME remesher)
set_target_properties(blitz_eval PROPERTIES EXPORT_NAME evaluator)
add_library(Blitz::remesher ALIAS blitzremesher)
add_library(Blitz::evaluator ALIAS blitz_eval)
if(BLITZ_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "CUDA architectures for neural kernels")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit 12 REQUIRED)
  find_package(OpenSSL REQUIRED)
  target_sources(blitzremesher PRIVATE src/neural_model_io.cpp src/neural_inference.cu src/neural_evaluate.cu)
  target_compile_definitions(blitzremesher PRIVATE BLITZ_CUDA)
  target_link_libraries(blitzremesher PRIVATE CUDA::cudart CUDA::cublas OpenSSL::Crypto)
  set_target_properties(blitzremesher PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(blitzremesher PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true>)
endif()
if(BLITZ_NEURAL_TRAIN AND NOT BLITZ_CUDA)
  message(FATAL_ERROR "BLITZ_NEURAL_TRAIN requires BLITZ_CUDA")
endif()
if(MSVC)
  target_compile_options(blitzremesher PRIVATE /W4)
  target_compile_options(blitz_eval PRIVATE /W4 /fp:precise)
else()
  target_compile_options(blitzremesher PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
  target_compile_options(blitz_eval PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
endif()
if(BLITZ_TOOLS)
  find_package(nlohmann_json REQUIRED)
  find_package(CURL REQUIRED)
  find_package(OpenSSL REQUIRED)
  find_package(LibArchive REQUIRED)
  add_library(blitz_io src/io.cpp)
  target_include_directories(blitz_io PUBLIC include PRIVATE third_party)
  target_link_libraries(blitz_io PUBLIC blitzremesher nlohmann_json::nlohmann_json)
  add_executable(blitz tools/main.cpp tools/benchmark.cpp)
  target_link_libraries(blitz PRIVATE blitz_io OpenSSL::Crypto)
  if(BLITZ_CUDA)
    add_executable(blitz-neural-prepare tools/neural_prepare.cpp)
    target_include_directories(blitz-neural-prepare PRIVATE src)
    target_link_libraries(blitz-neural-prepare PRIVATE blitz_io)
    add_executable(blitz-neural-action-prepare tools/neural_action_prepare.cpp)
    target_include_directories(blitz-neural-action-prepare PRIVATE src)
    target_link_libraries(blitz-neural-action-prepare PRIVATE blitz_io)
    add_executable(blitz-neural-action-proof tools/neural_action_proof.cpp)
    target_include_directories(blitz-neural-action-proof PRIVATE src)
    target_link_libraries(blitz-neural-action-proof PRIVATE blitz_io)
  endif()
  if(BLITZ_NEURAL_TRAIN)
    set(BLITZ_LIBTORCH_ROOT "$ENV{BLITZ_LIBTORCH_ROOT}" CACHE PATH "Official LibTorch shared-with-deps directory")
    if(NOT EXISTS "${BLITZ_LIBTORCH_ROOT}/include/torch/csrc/api/include/torch/torch.h")
      message(FATAL_ERROR "Set BLITZ_LIBTORCH_ROOT to the pinned LibTorch C++ distribution")
    endif()
    add_executable(blitz-neural-train tools/neural_train.cpp)
    add_executable(blitz-neural-replay tools/neural_replay.cpp)
    add_executable(blitz-neural-action-train tools/neural_action_train.cpp)
    foreach(neural_tool blitz-neural-train blitz-neural-replay blitz-neural-action-train)
    target_include_directories(${neural_tool} PRIVATE src)
    target_include_directories(${neural_tool} SYSTEM PRIVATE "${BLITZ_LIBTORCH_ROOT}/include" "${BLITZ_LIBTORCH_ROOT}/include/torch/csrc/api/include")
    target_compile_definitions(${neural_tool} PRIVATE _GLIBCXX_USE_CXX11_ABI=1)
    target_link_directories(${neural_tool} PRIVATE "${BLITZ_LIBTORCH_ROOT}/lib")
    # CUDA backend registration runs when this DSO loads. Ubuntu's --as-needed
    # can discard it because tensor calls dispatch through torch_cpu.
    target_link_libraries(${neural_tool} PRIVATE blitz_io CUDA::cudart torch torch_cpu
      "$<$<PLATFORM_ID:Linux>:-Wl,--push-state,--no-as-needed>" torch_cuda
      "$<$<PLATFORM_ID:Linux>:-Wl,--pop-state>" c10 c10_cuda)
    set_target_properties(${neural_tool} PROPERTIES BUILD_RPATH "${BLITZ_LIBTORCH_ROOT}/lib")
    endforeach()
  endif()
  add_executable(blitz-corpus tools/corpus.cpp)
  add_library(blitz_foliage_tools STATIC tools/foliage.cpp)
  target_include_directories(blitz_foliage_tools PRIVATE third_party)
  target_link_libraries(blitz_foliage_tools PUBLIC nlohmann_json::nlohmann_json PRIVATE CURL::libcurl OpenSSL::Crypto LibArchive::LibArchive)
  if(MSVC)
    target_compile_options(blitz_foliage_tools PRIVATE /fp:strict)
  else()
    target_compile_options(blitz_foliage_tools PRIVATE -ffp-contract=off -fno-fast-math)
  endif()
  target_link_libraries(blitz-corpus PRIVATE blitz_foliage_tools CURL::libcurl OpenSSL::Crypto LibArchive::LibArchive)
  add_executable(blitz-microbench tools/microbench.cpp)
  target_link_libraries(blitz-microbench PRIVATE blitzremesher nlohmann_json::nlohmann_json)
  add_executable(blitz-precision-bench tools/precision_bench.cpp)
  target_include_directories(blitz-precision-bench PRIVATE src)
  target_link_libraries(blitz-precision-bench PRIVATE blitzremesher nlohmann_json::nlohmann_json)
  add_executable(blitz-report tools/report.cpp)
  target_link_libraries(blitz-report PRIVATE nlohmann_json::nlohmann_json)
  add_executable(blitz-board tools/board.cpp)
  target_link_libraries(blitz-board PRIVATE nlohmann_json::nlohmann_json OpenSSL::Crypto)
  add_executable(blitz-probe tools/probe.cpp)
  target_link_libraries(blitz-probe PRIVATE blitz_io OpenSSL::Crypto)
  add_executable(blitz-lod-report tools/lod_report.cpp)
  target_link_libraries(blitz-lod-report PRIVATE blitz_io OpenSSL::Crypto)
  add_executable(blitz-tail-audit tools/tail_audit.cpp)
  target_include_directories(blitz-tail-audit PRIVATE src)
  target_link_libraries(blitz-tail-audit PRIVATE blitz_io OpenSSL::Crypto)
endif()

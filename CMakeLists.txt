cmake_minimum_required(VERSION 3.20)
project(BlitzRemesher VERSION 0.2.0 LANGUAGES C CXX)
include(CTest)
option(BLITZ_TOOLS "Build importer and corpus/research tools" ON)
option(BLITZ_RESEARCH "Build pinned external baselines" OFF)
option(BLITZ_SANITIZE "Address and undefined-behavior sanitizers" OFF)
option(BLITZ_FUZZ "Build libFuzzer targets (Clang)" OFF)
option(BLITZ_AVX2 "Enable runtime-dispatched AVX2 kernels" ON)
if(BLITZ_FLOAT_QUADRICS OR BLITZ_FLOAT_COSTS OR (DEFINED BLITZ_PACKED_COVERAGE AND NOT BLITZ_PACKED_COVERAGE))
  message(FATAL_ERROR "Experimental precision variants were removed. Production uses packed coverage and double quadrics/costs. See research/precision/ARCHIVE.md to restore the experiments.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(BLITZ_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(blitz_eval src/evaluate.cpp src/render_cost.cpp)
target_include_directories(blitz_eval PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_definitions(blitz_eval PRIVATE $<$<BOOL:${BLITZ_AVX2}>:BLITZ_AVX2>)
add_library(blitzremesher src/mesh.cpp src/simplify.cpp src/chain.cpp src/components.cpp src/c_api.cpp)
target_include_directories(blitzremesher PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(blitzremesher PUBLIC blitz_eval)
target_compile_features(blitzremesher PUBLIC cxx_std_20)
target_compile_features(blitz_eval PUBLIC cxx_std_20)
set_target_properties(blitzremesher blitz_eval PROPERTIES POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
set_target_properties(blitzremesher blitz_eval PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 2)
set_target_properties(blitzremesher PROPERTIES EXPORT_NAME remesher)
set_target_properties(blitz_eval PROPERTIES EXPORT_NAME evaluator)
add_library(Blitz::remesher ALIAS blitzremesher)
add_library(Blitz::evaluator ALIAS blitz_eval)
if(MSVC)
  target_compile_options(blitzremesher PRIVATE /W4)
  target_compile_options(blitz_eval PRIVATE /W4 /fp:precise)
else()
  target_compile_options(blitzremesher PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(blitz_eval PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
endif()
if(BLITZ_TOOLS)
  find_package(nlohmann_json REQUIRED)
  find_package(CURL REQUIRED)
  find_package(OpenSSL REQUIRED)
  find_package(LibArchive REQUIRED)
  add_library(blitz_io src/io.cpp)
  target_include_directories(blitz_io PUBLIC include PRIVATE third_party)
  target_link_libraries(blitz_io PUBLIC blitzremesher nlohmann_json::nlohmann_json)
  add_executable(blitz tools/main.cpp tools/benchmark.cpp)
  target_link_libraries(blitz PRIVATE blitz_io OpenSSL::Crypto)
  add_executable(blitz-corpus tools/corpus.cpp)
  add_library(blitz_foliage_tools STATIC tools/foliage.cpp)
  target_include_directories(blitz_foliage_tools PRIVATE third_party)
  target_link_libraries(blitz_foliage_tools PUBLIC nlohmann_json::nlohmann_json PRIVATE CURL::libcurl OpenSSL::Crypto LibArchive::LibArchive)
  if(MSVC)
    target_compile_options(blitz_foliage_tools PRIVATE /fp:strict)
  else()
    target_compile_options(blitz_foliage_tools PRIVATE -ffp-contract=off -fno-fast-math)
  endif()
  target_link_libraries(blitz-corpus PRIVATE blitz_foliage_tools CURL::libcurl OpenSSL::Crypto LibArchive::LibArchive)
  add_executable(blitz-microbench tools/microbench.cpp)
  target_link_libraries(blitz-microbench PRIVATE blitzremesher nlohmann_json::nlohmann_json)
  add_executable(blitz-precision-bench tools/precision_bench.cpp)
  target_include_directories(blitz-precision-bench PRIVATE src)
  target_link_libraries(blitz-precision-bench PRIVATE blitzremesher nlohmann_json::nlohmann_json)
  add_executable(blitz-report tools/report.cpp)
  target_link_libraries(blitz-report PRIVATE nlohmann_json::nlohmann_json)
  add_executable(blitz-board tools/board.cpp)
  target_link_libraries(blitz-board PRIVATE nlohmann_json::nlohmann_json OpenSSL::Crypto)
  add_executable(blitz-probe tools/probe.cpp)
  target_link_libraries(blitz-probe PRIVATE blitz_io OpenSSL::Crypto)
  add_executable(blitz-lod-report tools/lod_report.cpp)
  target_link_libraries(blitz-lod-report PRIVATE blitz_io OpenSSL::Crypto)
  add_executable(blitz-tail-audit tools/tail_audit.cpp)
  target_link_libraries(blitz-tail-audit PRIVATE blitz_io OpenSSL::Crypto)
endif()
if(BUILD_TESTING)
  add_executable(blitz-vegetation-tests tests/vegetation.cpp)
  target_include_directories(blitz-vegetation-tests PRIVATE src)
  target_link_libraries(blitz-vegetation-tests PRIVATE blitzremesher)
  add_test(NAME vegetation-contracts COMMAND blitz-vegetation-tests)
  add_executable(blitz-tests tests/contracts.cpp)
  target_link_libraries(blitz-tests PRIVATE blitzremesher)
  add_test(NAME contracts COMMAND blitz-tests)
  add_executable(blitz-precision-tests tests/precision.cpp)
  target_include_directories(blitz-precision-tests PRIVATE src)
  target_link_libraries(blitz-precision-tests PRIVATE blitzremesher)
  add_test(NAME precision-contracts COMMAND blitz-precision-tests)
  add_executable(blitz-render-cost-tests tests/render_cost.cpp)
  target_link_libraries(blitz-render-cost-tests PRIVATE blitzremesher)
  add_test(NAME render-cost-contracts COMMAND blitz-render-cost-tests)
  add_executable(blitz-c-smoke tests/c_smoke.c)
  target_link_libraries(blitz-c-smoke PRIVATE blitzremesher)
  set_target_properties(blitz-c-smoke PROPERTIES LINKER_LANGUAGE CXX)
  add_test(NAME c-abi COMMAND blitz-c-smoke)
  if(BLITZ_TOOLS)
    add_executable(blitz-foliage-tests tests/foliage.cpp)
    target_link_libraries(blitz-foliage-tests PRIVATE blitz_foliage_tools LibArchive::LibArchive)
    add_test(NAME foliage-contracts COMMAND blitz-foliage-tests)
    add_executable(blitz-io-tests tests/io_tests.cpp)
    target_link_libraries(blitz-io-tests PRIVATE blitz_io)
    add_test(NAME io-contracts COMMAND blitz-io-tests)
  endif()
endif()
if(BLITZ_RESEARCH)
  include(cmake/Baselines.cmake)
endif()
if(BLITZ_FUZZ)
  add_executable(blitz-fuzz tests/fuzz.cpp src/mesh.cpp src/simplify.cpp)
  target_include_directories(blitz-fuzz PRIVATE include)
  target_compile_options(blitz-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(blitz-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS blitzremesher blitz_eval EXPORT BlitzRemesherTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/blitz/blitz.h include/blitz/mesh.hpp include/blitz/evaluate.hpp include/blitz/render_cost.hpp include/blitz/remesher.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/blitz)
install(EXPORT BlitzRemesherTargets NAMESPACE Blitz:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BlitzRemesher)
configure_package_config_file(cmake/BlitzRemesherConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/BlitzRemesherConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BlitzRemesher)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/BlitzRemesherConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/BlitzRemesherConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/BlitzRemesherConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BlitzRemesher)
